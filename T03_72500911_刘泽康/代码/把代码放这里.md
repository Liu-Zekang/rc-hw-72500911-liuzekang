def count_colors(dets):
    """循环+字典：统计各类目标数量
    dets: [(x,y,w,h,conf,cls_name),...]
    return {类别名:数量}
    """
    cnt = {}
    for box in dets:
        cls_name = box[-1]
        if cls_name in cnt:
            cnt[cls_name] += 1
        else:
            cnt[cls_name] = 1
    return cnt


def largest_box(dets):
    """遍历列表，找出面积最大检测框，空列表返回None"""
    if not dets:
        return None
    max_area = -1
    max_box = None
    for b in dets:
        _, _, w, h, _, _ = b
        area = w * h
        if area > max_area:
            max_area = area
            max_box = b
    return max_box


def filter_by_conf(dets, threshold):
    """列表推导式，过滤置信度低于threshold的框"""
    return [box for box in dets if box[4] >= threshold]


def clamp(v, lo, hi):
    """边界限幅，v小于lo返回lo，大于hi返回hi，否则返回v"""
    if v < lo:
        return lo
    elif v > hi:
        return hi
    else:
        return v


def moving_average(window, new_val, win_size=3):
    window.append(new_val)
    if len(window) > win_size:
        window.pop(0)
    avg = sum(window) / len(window)
    return avg


def parse_detection_line(line):
    """字符串切割+类型转换，解析一行检测文本
    输入样例："10 20 50 60 0.85 person"
    返回元组 (10,20,50,60,0.85,"person")
    """
    parts = line.strip().split()
    x = int(parts[0])
    y = int(parts[1])
    w = int(parts[2])
    h = int(parts[3])
    conf = float(parts[4])
    cls = parts[5]
    return (x, y, w, h, conf, cls)
