def detect_color(frame, color_name):
    """在 BGR 帧里找出 color_name 对应颜色目标的质心。
    :param frame: BGR 图（单帧）
    :param color_name: "red" / "blue" / "green" / "yellow"
    :return: (cx, cy) 或 None（没找到）
    提示：转 HSV → inRange（红色两段合并）→ 形态学开+闭 → 最大轮廓 → moments
    """
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
    mask = None
    if color_name == "red":
        # 红色HSV两段区间
        lower_red1 = np.array([0, 100, 80])
        upper_red1 = np.array([10, 255, 255])
        lower_red2 = np.array([170, 100, 80])
        upper_red2 = np.array([180, 255, 255])
        mask1 = cv2.inRange(hsv, lower_red1, upper_red1)
        mask2 = cv2.inRange(hsv, lower_red2, upper_red2)
        mask = cv2.bitwise_or(mask1, mask2)
    elif color_name == "blue":
        lower = np.array([90, 100, 80])
        upper = np.array([130, 255, 255])
        mask = cv2.inRange(hsv, lower, upper)
    elif color_name == "green":
        lower = np.array([40, 100, 80])
        upper = np.array([75, 255, 255])
        mask = cv2.inRange(hsv, lower, upper)
    elif color_name == "yellow":
        lower = np.array([20, 100, 80])
        upper = np.array([35, 255, 255])
        mask = cv2.inRange(hsv, lower, upper)
    else:
        return None
    # 形态学：先开后闭，去噪填充小洞
    kernel = np.ones((5, 5), np.uint8)
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
    # 找轮廓
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    if len(contours) == 0:
        return None
    # 取面积最大轮廓
    max_contour = max(contours, key=cv2.contourArea)
    M = cv2.moments(max_contour)
    if M["m00"] == 0:
        return None
    cx = M["m10"] / M["m00"]
    cy = M["m01"] / M["m00"]
    return (cx, cy)

def open_source(src):
    """打开视频源并做低延迟设置，返回 cv2.VideoCapture（打不开也要返回对象）。
    src 可能是 0（摄像头）、"http://..."（网络流）或一个视频文件路径。
    """
    cap = cv2.VideoCapture(src)
    # 设置缓冲区大小=1，降低延迟
    cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)
    return cap

def main():
    current_color = "red"
    fps = FPSMeter()
    cap = open_source(0)
    while True:
        ret, frame = cap.read()
        if not ret:
            frame = FALLBACK_IMAGE.copy()
        fps.tick()
        center = detect_color(frame, current_color)
        # BGR颜色
        color_map = {
            "red": (0,0,255),
            "blue": (255,0,0),
            "green": (0,255,0),
            "yellow": (0,255,255)
        }
        if center is not None:
            cx, cy = int(center[0]), int(center[1])
            cv2.drawMarker(frame, (cx, cy), color_map[current_color], markerType=cv2.MARKER_CROSS, thickness=2)
        # HUD文字，fps.fps不带括号
        put_hud(frame, f"Color: {current_color} | FPS: {fps.fps:.1f}")
        cv2.imshow("Color Tracker", frame)
        key = cv2.waitKey(1) & 0xFF
        if key == ord('1'):
            current_color = "red"
            print("切换追踪：红色")
        elif key == ord('2'):
            current_color = "blue"
            print("切换追踪：蓝色")
        elif key == ord('3'):
            current_color = "green"
            print("切换追踪：绿色")
        elif key == ord('4'):
            current_color = "yellow"
            print("切换追踪：黄色")
        elif key == ord('q'):
            break
    cap.release()
    cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
