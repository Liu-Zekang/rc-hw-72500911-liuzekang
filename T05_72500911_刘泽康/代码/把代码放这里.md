import cv2
import numpy as np
from typing import Optional, Tuple
import vision_util as vu
# 函数1：读取图像（替代 vision_util.imread）
def imread(path: str) -> np.ndarray:
    return cv2.imread(path)

# 函数2：彩色图像转灰度图
def to_grayscale(img: np.ndarray) -> np.ndarray:
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    return gray

# 函数3：灰度图像二值化
def binarize(gray_img: np.ndarray, thresh: int = 127) -> np.ndarray:
    _, binary_img = cv2.threshold(gray_img, thresh, 255, cv2.THRESH_BINARY)
    return binary_img

# 函数4：基于HSV生成颜色掩膜
def hsv_mask(img_bgr: np.ndarray, lower_hsv: Tuple[int, int, int], upper_hsv: Tuple[int, int, int]) -> np.ndarray:
    hsv_img = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2HSV)
    mask = cv2.inRange(hsv_img, lower_hsv, upper_hsv)
    return mask

# 函数5：读取并播放视频
def read_frames(video_path: str) -> None:
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        print("视频源打开失败！")
        raise SystemExit(1)
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        cv2.imshow("video", frame)
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
    cap.release()
    cv2.destroyAllWindows()

# 函数6：计算掩膜白色区域质心
def mask_centroid(mask: np.ndarray) -> Optional[Tuple[float, float]]:
    moments = cv2.moments(mask)
    if moments["m00"] == 0:
        return None
    cx = moments["m10"] / moments["m00"]
    cy = moments["m01"] / moments["m00"]
    return (cx, cy)


# ===================== 作业主程序：视频追踪 =====================
if __name__ == "__main__":
    import time
    # 读取 patterns.png 作为模拟视频帧
    img = cv2.imread("patterns.png")
    if img is None:
        print("图片读取失败，请确认 patterns.png 和 hw_opencv.py 在同一目录！")
        exit()

    # 黄色小球HSV阈值（你之前验证可用）
    lower = (20, 100, 100)
    upper = (30, 255, 255)

    while True:
        frame = img.copy()
        mask = hsv_mask(frame, lower, upper)
        center = mask_centroid(mask)

        if center is not None:
            cx, cy = center
            # 绘图，优先用vision_util，如果报错就换成 cv2.circle
            try:
                vu.draw_point(frame, (int(cx), int(cy)))
            except:
                cv2.circle(frame, (int(cx), int(cy)), 6, (0,0,255), -1)

            print(f"小球质心坐标：x={cx:.2f}, y={cy:.2f}")

        cv2.imshow("video", frame)
        cv2.imshow("mask", mask)

        # 按 q 键退出循环
        if cv2.waitKey(20) & 0xFF == ord('q'):
            break
        time.sleep(0.02)

    cv2.destroyAllWindows()
