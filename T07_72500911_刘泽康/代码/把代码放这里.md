from ultralytics import YOLO

# COCO类别名称映射
coco_names = {
    0: "person",
    5: "bus",
    11: "stop sign"
}

model = YOLO("yolov8n.pt")
results = model("bus.jpg")

for result in results:
    boxes = result.boxes
    print("===== 检测结果 =====")
    print("检测框坐标(xyxy):\n", boxes.xyxy)
    print("置信度:\n", boxes.conf)
    cls_ids = boxes.cls.int().tolist()
    cls_names = [coco_names.get(id, f"未知{id}") for id in cls_ids]
    print("类别名称：", cls_names)

result.save("bus_result_new.jpg")
